\begin{frame}{Il problema e la domanda}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Il problema}
        \small
        I motori di ricerca ottimizzano la \textbf{pertinenza}, non la
        \textbf{diversità}. Su un tema divisivo la query porta con sé
        un'inclinazione, e il ranking la amplifica.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{La domanda}
        \small
        Si può \textbf{misurare} quanto un ecosistema informativo sia omogeneo
        nelle opinioni, partendo dai documenti stessi?
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{itemize}\small
    \item \textbf{Intervento}: i risultati non sono una lista, sono \alert{tre colonne}
          FAVOR / NEUTRAL / AGAINST affiancate.
    \item \textbf{Misura}: la \alert{Stance Diversity} --- entropia di Shannon normalizzata
          sulle stance dei risultati.
    \item \textbf{Confronto}: la stessa pipeline su \alert{due ecosistemi} (news e social),
          senza differenze di trattamento.
  \end{itemize}
  \vspace{0.3em}
  {\footnotesize\color{echoslate} Temi: energia nucleare · cambiamento climatico · vaccini.}
\end{frame}

\begin{frame}{Motore di ricerca: due stadi}
  \centering
  \resizebox{\textwidth}{!}{%
  \begin{tikzpicture}[node distance=0.7cm]
    \node[io, minimum width=2.1cm] (c) {Corpus\\\footnotesize $\sim$800 doc};
    \node[box, right=of c, minimum width=3.5cm] (s1)
      {\textbf{Stadio 1} --- recupero\\\footnotesize BM25 $\oplus$ bge+FAISS, min-max\\\footnotesize $\alpha{=}0{,}45$, $\beta{=}0{,}55$};
    \node[box, right=of s1, minimum width=3.3cm] (s2)
      {\textbf{Stadio 2} --- pertinenza\\\footnotesize bge-reranker (cross-encoder)};
    \node[io, right=of s2, minimum width=1.6cm] (o) {Top-$k$};
    \draw[arr] (c) -- (s1); \draw[arr] (s1) -- node[lbl, above]{50} (s2);
    \draw[arr] (s2) -- node[lbl, above]{30} (o);
  \end{tikzpicture}}
  \vspace{0.7em}
  \begin{itemize}\small
    \item \textbf{Complementari, non alternativi}: la fusione \emph{recupera}, il reranker
          \emph{riordina}. Il tetto del reranker è il recall dello stadio 1.
    \item \textbf{BM25 con $b = 0{,}60$} (non 0,75): su articoli tutti lunghi, penalizzare la
          lunghezza scarta proprio i pezzi ricchi di evidenze.
    \item \textbf{bge è asimmetrico}: prefisso-istruzione solo sulle query, mai sui passaggi.
    \item Aggregazione \textbf{max-pooling} sui chunk: un documento è rilevante se
          \emph{almeno un} suo chunk lo è.
  \end{itemize}
\end{frame}

\begin{frame}{Qrels: protocollo e accordo fra annotatori}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \footnotesize
      \textbf{Pool alla Cranfield}, unione di \alert{5 run}: BM25, denso, RRF, fusione,
      pool del reranker. Profondità 20 $\Rightarrow$ \textbf{380} coppie (query, doc).\\[0.4em]
      \textbf{143 su 380 --- il 38\% --- vengono da un solo run}: pescare dalla sola
      fusione le avrebbe rese non giudicate, cioè non rilevanti, falsando l'ablazione a
      sfavore delle baseline.\\[0.4em]
      Ordine randomizzato, rango e punteggio \alert{nascosti} all'annotatore
      (\emph{position bias}).\\[0.4em]
      \textbf{Doppia annotazione umana in cieco}: fogli separati, nessuna consultazione
      durante il lavoro --- l'accordo su annotazioni allineate strada facendo non
      significherebbe nulla.
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering\small
      \begin{tabular}{lccc}
        \toprule
        \textbf{Tema} & $n$ & Accordo & $\boldsymbol{\kappa}$ \textbf{quad.} \\
        \midrule
        Nucleare & 137 & 0,839 & \textbf{0,904} \\
        Clima    & 109 & 0,761 & \textbf{0,814} \\
        Vaccini  & 134 & 0,784 & \textbf{0,856} \\
        \midrule
        \textbf{Totale} & \textbf{380} & 0,797 & \textbf{0,867} \\
        \bottomrule
      \end{tabular}\\[0.5em]
      \footnotesize
      \emph{Almost perfect} su tutti i temi (non pesato: 0,675). Scala ordinale $\Rightarrow$
      $\kappa$ a \textbf{pesi quadratici}: quello non pesato tratterebbe ``0 vs 1'' e ``0 vs 2''
      come lo stesso errore.\\[0.3em]
      {\scriptsize\color{echoslate} 77 divergenze su 380, di cui \alert{2 a due gradini}: si
      discute \emph{quanto} un documento serva, non \emph{se} risponda.}
    \end{column}
  \end{columns}
  \vspace{0.3em}
  {\footnotesize\color{echogreen}\textbf{Controllo di robustezza:} le due serie ordinano allo stesso
  modo \textbf{5 sistemi su 6} ($\tau_b$ di Kendall $=1{,}0$); sui 6 scende a $0{,}733$ e l'unico a
  spostarsi è la \emph{fusione tarata} --- per $0{,}0006$ di NDCG, cioè proprio il sistema che
  concludiamo non servire.}
\end{frame}

\begin{frame}{Stance Diversity --- la metrica firma}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small
      \[
        \text{Div}(p) = -\frac{1}{\log 3}\!\!\sum_{c \in \{\text{F},\text{N},\text{A}\}}\!\! p_c \log p_c \;\in\; [0,1]
      \]
      \vspace{-0.4em}
      \begin{itemize}\footnotesize
        \item Calcolata sull'\alert{intero pool} riordinato, \textbf{non} sulle colonne
              mostrate: le colonne sono bilanciate per costruzione, misurarle darebbe
              una costante travestita da misura.
        \item Due numeri affiancati nella GUI: la barra = \emph{ciò che il corpus offre},
              le colonne = \emph{ciò che il sistema mostra}.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering\scriptsize
      \begin{tabular}{lcl}
        \toprule
        (F,\,N,\,A) & Div & lettura \\
        \midrule
        $(30,0,0)$   & 0,00 & echo chamber perfetta \\
        $(28,1,1)$   & 0,27 & quasi unanime \\
        $(20,8,2)$   & 0,73 & minoranza presente \\
        $(15,10,5)$  & 0,92 & sbilanciato ma reale \\
        $(10,10,10)$ & 1,00 & bilanciamento perfetto \\
        \bottomrule
      \end{tabular}\\[0.5em]
      \footnotesize\color{echoslate}
      Invariante per permutazione delle classi e insensibile alla dimensione del
      campione: per questo l'interfaccia mostra \emph{sempre} anche la distribuzione.
    \end{column}
  \end{columns}
\end{frame}

  \end{tabular}
  \vspace{0.6em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{\small Il risultato solido}
        \scriptsize
        L'ecosistema \textbf{news} è marcatamente omogeneo \emph{in senso assoluto}, e due
        misure indipendenti concordano: diversity $0{,}23$--$0{,}55$ sul pool, e
        $\approx$\textbf{3\%} di documenti AGAINST sull'intero corpus.
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{\small Il confronto, invece, è confondato}
        \scriptsize
        Il social ha entropia a priori $1{,}000$ \emph{per costruzione}, le news quella
        naturale. La differenza somma un effetto reale ignoto e un artefatto di
        campionamento di segno noto (positivo): \textbf{sovrastima} la conclusione.
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.3em}
  {\footnotesize\textbf{Contromisura proposta:} \emph{diversity lift} $=
  H(p_{\text{pool}})/H(p_{\text{corpus}})$, invariante rispetto alla distribuzione a priori.
  Definita, non ancora calcolata.}
\end{frame}

    \node[box, right=of ev, minimum width=2.4cm, fill=echored!8, draw=echored] (gate)
      {\textbf{Cancello}\\\footnotesize coseno $\ge 0{,}62$};
    \node[box, above right=0.05cm and 0.8cm of gate, minimum width=2.3cm] (nli)
      {\textbf{NLI}\\\footnotesize DeBERTa-fever};
    \node[box, below right=0.05cm and 0.8cm of gate, minimum width=2.3cm] (llm)
      {\textbf{LLM}\\\footnotesize qwen2.5:7b};
    \draw[arr] (cl) -- (ev); \draw[arr] (ev) -- (gate);
    \draw[arr] (gate.east) -- (nli.west); \draw[arr] (gate.east) -- (llm.west);
  \end{tikzpicture}
  \vspace{0.5em}
  \begin{itemize}\small
    \item \textbf{Prime due fasi condivise, terza intercambiabile}: cambia \emph{solo} il
          motore di verdetto $\Rightarrow$ la differenza misurata è attribuibile ad esso.
    \item \textbf{Il cancello di pertinenza} evita i falsi REFUTED: su una KB tematica il
          retrieval restituisce \emph{sempre} qualcosa, e l'NLI legge ``diverso'' come
          ``contraddittorio''.
    \item \textbf{Grounding puro} per l'LLM: giudica solo sulle evidenze, mai sulla
          conoscenza parametrica. Tutto locale, nessuna API.
  \end{itemize}
\end{frame}

\begin{frame}{Limiti dichiarati e prossimi passi}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{\small Limiti}
        \scriptsize
        \begin{itemize}\scriptsize
          \item \textbf{Confronto news/social confondato} dal campionamento stratificato.
          \item \textbf{11 query}: bastano a ordinare i sistemi, non a tarare $\alpha$ né a
                dare intervalli di confidenza.
          \item \textbf{Qrels non riconciliati}: le 77 divergenze fra i due annotatori non
                sono state adjudicate in un'unica serie.
          \item NEUTRAL è un costrutto instabile fuori dai benchmark.
          \item KB statica: claim recenti $\to$ evidenza insufficiente.
          \item Summarization non valutata.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{\small Prossimi passi}
        \scriptsize
        \begin{itemize}\scriptsize
          \item \textbf{Diversity lift}: chiude il confondimento principale, costa una
                passata offline dello stance detector.
          \item \textbf{Più query} (40--50) per intervalli di confidenza e test di
                significatività.
          \item \textbf{Adjudicare le divergenze} fra i due annotatori in una serie
                consolidata.
          \item \textbf{Combinare i due motori} di fact-check: i profili di errore sono
                complementari.
          \item Persistenza dell'indice KB; parallelizzazione del ramo fact-check.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
